cmake_minimum_required(VERSION 3.18)
project(homeimg_desktop LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
find_package(nlohmann_json REQUIRED)
find_package(BLAS REQUIRED)
add_executable(homeimg_sample ../../tools/homeimg_sample.cpp)
target_compile_features(homeimg_sample PRIVATE cxx_std_17)
target_compile_definitions(homeimg_sample PRIVATE EIGEN_USE_BLAS)
target_link_libraries(homeimg_sample PRIVATE Eigen3::Eigen nlohmann_json::nlohmann_json BLAS::BLAS)
add_executable(check_homeimg_velocity EXCLUDE_FROM_ALL ../../tools/check_homeimg_velocity.cpp)
target_link_libraries(check_homeimg_velocity PRIVATE Eigen3::Eigen nlohmann_json::nlohmann_json BLAS::BLAS)
target_compile_features(check_homeimg_velocity PRIVATE cxx_std_17)
target_compile_definitions(check_homeimg_velocity PRIVATE EIGEN_USE_BLAS)
add_library(homeimg_cpu SHARED homeimg.cpp)
target_compile_features(homeimg_cpu PRIVATE cxx_std_17)
target_compile_definitions(homeimg_cpu PRIVATE EIGEN_USE_BLAS)
target_link_libraries(homeimg_cpu PRIVATE Eigen3::Eigen nlohmann_json::nlohmann_json BLAS::BLAS)
